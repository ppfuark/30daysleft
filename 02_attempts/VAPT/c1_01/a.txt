def arrow(c, x0, y0, x1, y1, col=OUT, w=8, both=False):
    c.line([(x0, y0), (x1, y1)], fill=col, w=w)
    a = math.atan2(y1 - y0, x1 - x0)
    def head(x, y, a):
        c.poly([(x, y), (x - 26 * math.cos(a - .45), y - 26 * math.sin(a - .45)), (x - 26 * math.cos(a + .45), y - 26 * math.sin(a + .45))], fill=col, ol=col, w=2)
    head(x1, y1, a)
    if both: head(x0, y0, a + math.pi)


def flood(c, t, n, xa, xb, ya, yb, seed=1, size=18):
    cols = [RED, BLUE, GREEN, YEL, PURPLE, (255, 140, 40)]
    for i in range(n):
        ph = (t * .6 + i * .618 + seed * .13) % 1
        x = lerp(xa, xb, ph); y = lerp(ya, yb, (i * .37 + seed * .21) % 1) + math.sin(t * 4 + i) * 8
        col = cols[i % 6]
        if i % 3 == 0: c.ell(x, y, size / 2, size / 2, fill=col, w=2)
        elif i % 3 == 1: c.rect(x - size / 2, y - size / 2, x + size / 2, y + size / 2, fill=col, r=3, w=2)
        else: c.poly([(x, y - size / 2), (x + size / 2, y + size / 2), (x - size / 2, y + size / 2)], fill=col, w=2)

def stamp(c, x, y, txt, size, ang, k=1.0, col=RED):
    lay = Image.new("RGBA", (W * S, 200 * S), (0, 0, 0, 0))
    d = ImageDraw.Draw(lay)
    f = font(P(size * k))
    bb = d.textbbox((0, 0), txt, font=f, anchor="lt")
    tw, th = bb[2] - bb[0], bb[3] - bb[1]
    cx, cy = W * S // 2, 100 * S
    pad = P(14)
    d.rounded_rectangle([cx - tw // 2 - pad, cy - th // 2 - pad, cx + tw // 2 + pad, cy + th // 2 + pad], radius=P(12), outline=col + (255,), width=P(10 * k))
    d.text((cx, cy), txt, font=f, fill=col + (255,), anchor="mm")
    lay = lay.rotate(ang, resample=Image.BICUBIC, expand=False, center=(cx, cy))
    c.im.paste(lay, (P(x) - cx, P(y) - cy), lay)


def grid(c, x, y, n=6, cell=64, hl=None):
    x0 = x - n * cell / 2; y0 = y - n * cell / 2
    for i in range(n):
        for j in range(n):
            col = (255, 255, 255) if (i + j) % 2 == 0 else (225, 235, 250)
            if hl:
                col = hl(i, j) or col
            c.rect(x0 + j * cell, y0 + i * cell, x0 + (j + 1) * cell, y0 + (i + 1) * cell, fill=col, w=2)
    return x0, y0

def silences(dur):
    out = subprocess.run(["ffmpeg", "-hide_banner", "-i", MP3, "-af", "silencedetect=noise=-35dB:d=0.25", "-f", "null", "-"], capture_output=True, text=True).stderr
    st = [float(x) for x in re.findall(r"silence_start: ([\d.]+)", out)]
    en = [float(x) for x in re.findall(r"silence_end: ([\d.]+)", out)]
    sil = []
    for i, s0 in enumerate(st):
        sil.append((s0, en[i] if i < len(en) else dur))
    return sil


def align(dur):
    sil = silences(dur)
    wt = [len(w) + (3 if w[-1] in ".?!:" else 1.5 if w[-1] == "," else 0) for w in WORDS]
    t0s = [0.0] * len(WORDS); t1s = [0.0] * len(WORDS)
    sp0 = sil[0][1] if sil and sil[0][0] < .05 else 0.0
    tail = dur
    if sil and sil[-1][1] >= dur - .05: tail = sil[-1][0]; sil = sil[:-1]
    rate = sum(wt) / (tail - sp0 - sum(b - a for a, b in sil if a > sp0))
    cur_t = sp0; cur_i = 0; sidx = 0
    def place(i0, i1, a, b):
        tot = sum(wt[i0:i1 + 1]); acc = a
        for k in range(i0, i1 + 1):
            d = (b - a) * wt[k] / tot
            t0s[k] = acc; t1s[k] = acc + d * .92; acc += d
    for i, w in enumerate(WORDS[:-1]):
        if w[-1] not in ".?!:,": continue
        pred = cur_t + sum(wt[cur_i:i + 1]) / rate
        thr = 1.2 if w[-1] in ".?!:" else .7
        best = None
        for j in range(sidx, len(sil)):
            if sil[j][0] < cur_t + .15: continue
            dd = abs(sil[j][0] - pred)
            if best is None or dd < best[0]: best = (dd, j)
            if sil[j][0] > pred + thr + .5: break
        if best and best[0] < thr:
            j = best[1]
            place(cur_i, i, cur_t, sil[j][0])
            cur_t = sil[j][1]; cur_i = i + 1; sidx = j + 1
    place(cur_i, len(WORDS) - 1, cur_t, tail)
    return t0s, t1s

W, H, S = 720, 1280, 2

_F = {}


def P(v): return int(round(v * S))
def lerp(a, b, u): return a + (b - a) * u
def clamp(v, lo, hi): return max(lo, min(hi, v))
def ease(u):
    u = clamp(u, 0, 1); return u * u * (3 - 2 * u)
def pop(t, d=.35):
    if t <= 0: return 0.0
    u = min(1.0, t / d); c1 = 1.70158; c3 = c1 + 1
    return 1 + c3 * (u - 1) ** 3 + c1 * (u - 1) ** 2


def font(px):
    px = max(4, int(round(px)))
    if px not in _F: _F[px] = ImageFont.truetype(FONT_PATH, px)
    return _F[px]


def gray(col):
    g = int(.3 * col[0] + .59 * col[1] + .11 * col[2]); return (g, g, g)


class Cv:
    def __init__(s, bg):
        s.im = Image.new("RGB", (W * S, H * S), bg)
        s.d = ImageDraw.Draw(s.im)

    def line(s, pts, fill=OUT, w=6):
        if fill is None: return
        q = [(P(x), P(y)) for x, y in pts]
        s.d.line(q, fill=fill, width=max(1, P(w)), joint="curve")
        r = P(w) / 2
        for x, y in (q[0], q[-1]):
            s.d.ellipse([x - r, y - r, x + r, y + r], fill=fill)

    def poly(s, pts, fill=None, ol=OUT, w=5):
        q = [(P(x), P(y)) for x, y in pts]
        if fill is not None: s.d.polygon(q, fill=fill)
        if w > 0 and ol is not None:
            s.d.line(q + [q[0], q[1]], fill=ol, width=max(1, P(w)), joint="curve")

    def ell(s, x, y, rx, ry, fill=None, ol=OUT, w=5):
        rx, ry = abs(rx), abs(ry)
        if rx < .3 or ry < .3: return
        s.d.ellipse([P(x - rx), P(y - ry), P(x + rx), P(y + ry)], fill=fill,
                    outline=ol if (w > 0 and ol is not None) else None, width=max(1, P(w)))

    def rect(s, x0, y0, x1, y1, fill=None, ol=OUT, r=0, w=5):
        if x1 < x0: x0, x1 = x1, x0
        if y1 < y0: y0, y1 = y1, y0
        if x1 - x0 < .5 or y1 - y0 < .5: return
        r = clamp(r, 0, min((x1 - x0) / 2, (y1 - y0) / 2))
        s.d.rounded_rectangle([P(x0), P(y0), P(x1), P(y1)], radius=P(r), fill=fill,
                              outline=ol if (w > 0 and ol is not None) else None, width=max(1, P(w)))

    def arc(s, cx, cy, rx, ry, a0, a1, fill=OUT, w=5):
        if rx < .5 or ry < .5: return
        s.d.arc([P(cx - rx), P(cy - ry), P(cx + rx), P(cy + ry)], a0, a1, fill=fill, width=max(1, P(w)))

    def text(s, x, y, txt, size, fill=OUT, sw=0, anchor="mm"):
        if size < 2: return
        s.d.text((P(x), P(y)), txt, font=font(P(size)), fill=fill, anchor=anchor,
                 stroke_width=P(sw), stroke_fill=OUT)

    c.poly([(x, y - size), (x + q, y - q), (x + size, y), (x + q, y + q), (x, y + size), (x - q, y + q), (x - size, y), (x - q, y - q)], fill=col, w=3)


def chip(c, x, y, label, col, w=120, h=40, size=20):
    while tw(label, size) > w - 16 and size > 8: size -= 1
    c.rect(x - w / 2, y - h / 2, x + w / 2, y + h / 2, fill=col, r=h / 2, w=4)
    c.text(x, y + 1, label, size, fill=WHITE, sw=1.5)


def banner(c, x, y, txt, size, col, tc=WHITE, sc=1.0):
    if sc < .05: return
    size = size * sc
    while tw(txt, size) > 680 and size > 10: size -= 1
    w = tw(txt, size) + 44 * sc; h = size * 1.75
    c.rect(x - w / 2, y - h / 2, x + w / 2, y + h / 2, fill=col, r=h * .28, w=5)
    c.text(x, y + size * .04, txt, size, fill=tc, sw=0 if tc == OUT else max(1.5, size * .05))


def cross(c, x, y, size, t):
    k = pop(t, .3)
    if k <= 0: return
    r = size * k; w = max(10, size * .22)
    for col, ww in ((OUT, w + 10), (RED, w)):
        c.line([(x - r, y - r), (x + r, y + r)], fill=col, w=ww)
        c.line([(x - r, y + r), (x + r, y - r)], fill=col, w=ww)


def check(c, x, y, r, t):
    k = pop(t, .3)
    if k <= 0: return
    R = r * k
    c.ell(x, y, R, R, fill=GREEN, w=max(2, R * .12))
    pts = [(x - .45 * R, y + .02 * R), (x - .12 * R, y + .38 * R), (x + .5 * R, y - .35 * R)]
    c.line(pts, fill=OUT, w=R * .34); c.line(pts, fill=WHITE, w=R * .22)


def drop(c, x, y, r):
    R = r; cy = y + .25 * R; d = 1.75 * R; al = math.acos(R / d)
    pts = [(x, cy - d)]
    n = 24
    for i in range(n + 1):
        ph = al + (2 * math.pi - 2 * al) * i / n
        pts.append((x + R * math.sin(ph), cy - R * math.cos(ph)))
    c.poly(pts, fill=PURPLE, w=max(2, r * .14))
    c.ell(x - .35 * R, cy + .1 * R, .16 * R, .26 * R, fill=(230, 190, 255), w=0)


def gear(c, x, y, r, t):
    pts = []
    n = 8
    for i in range(n * 4):
        a = t * 1.5 + 2 * math.pi * i / (n * 4)
        rr = r if (i % 4) in (0, 1) else r * .74
        pts.append((x + rr * math.cos(a), y + rr * math.sin(a)))
    c.poly(pts, fill=(150, 160, 185), w=max(2, r * .08))
    c.ell(x, y, r * .3, r * .3, fill=(70, 80, 110), w=max(2, r * .08))


def rays(c, x, y, r0, r1, n, t, col=(52, 74, 160)):
    R = 1500
    for i in range(n):
        a = t * .6 + 2 * math.pi * i / n; b = a + math.pi / n * .8
        c.poly([(x, y), (x + R * math.cos(a), y + R * math.sin(a)), (x + R * math.cos(b), y + R * math.sin(b))], fill=col, w=0)


def wifi(c, x, y, s=1.0, broken=False):
    col = (130, 135, 150) if broken else BLUE
    for r in (40, 75, 110):
        c.arc(x, y + 60 * s, r * s, r * s, 228, 312, fill=OUT, w=26 * s)
        c.arc(x, y + 60 * s, r * s, r * s, 228, 312, fill=col, w=16 * s)
    c.ell(x, y + 60 * s, 12 * s, 12 * s, fill=col, w=4)
    if broken:
        c.line([(x - 70 * s, y - 50 * s), (x + 70 * s, y + 80 * s)], fill=OUT, w=26 * s)
        c.line([(x - 70 * s, y - 50 * s), (x + 70 * s, y + 80 * s)], fill=RED, w=16 * s)


def envelope(c, x, y, w, h):
    if w < 6 or h < 6: return
    c.rect(x - w / 2, y - h / 2, x + w / 2, y + h / 2, fill=(255, 250, 235), r=10, w=5)
    c.line([(x - w / 2 + 6, y - h / 2 + 6), (x, y + h * .12), (x + w / 2 - 6, y - h / 2 + 6)], w=5)
    c.line([(x - w / 2 + 6, y + h / 2 - 6), (x - w * .12, y + h * .02)], w=4)
    c.line([(x + w / 2 - 6, y + h / 2 - 6), (x + w * .12, y + h * .02)], w=4)
    c.ell(x, y + h * .12, h * .12, h * .12, fill=RED, w=4)


def chef_hat(c, x, y, s=1.0):
    c.rect(x - 38 * s, y - 14 * s, x + 38 * s, y + 10 * s, fill=WHITE, r=6 * s, w=4)
    for dx, dy, r in ((-26, -34, 22), (0, -48, 27), (26, -34, 22)):
        c.ell(x + dx * s, y + dy * s, r * s, r * s, fill=WHITE, w=4)
    c.rect(x - 36 * s, y - 20 * s, x + 36 * s, y + 8 * s, fill=WHITE, r=6 * s, w=0)
    c.line([(x - 38 * s, y + 10 * s), (x + 38 * s, y + 10 * s)], w=4)


def card(c, x, y, w, h, idx, glow=0):
    if w < 10 or h < 10: return
    pal = [(150, 70, 210), (220, 60, 70), (40, 140, 200), (60, 170, 90), (240, 150, 40), (200, 80, 150), (70, 90, 200), (120, 120, 130), (30, 150, 150), (180, 60, 60)]
    col = pal[idx % len(pal)]
    if glow:
        for k in (26, 16, 8):
            c.rect(x - w / 2 - k, y - h / 2 - k, x + w / 2 + k, y + h / 2 + k, fill=(255, 235, 140), r=22, w=0)
    c.rect(x - w / 2, y - h / 2, x + w / 2, y + h / 2, fill=(245, 215, 110), r=w * .1, w=max(3, w * .03))
    c.rect(x - w * .42, y - h * .40, x + w * .42, y + h * .32, fill=col, r=w * .07, w=max(2, w * .025))
    # silhouette
    c.ell(x, y - h * .12, w * .18, w * .18, fill=(255, 230, 200), w=max(2, w * .02))
    c.rect(x - w * .24, y + h * .04, x + w * .24, y + h * .28, fill=(250, 250, 255), r=w * .06, w=max(2, w * .02))
    c.rect(x - w * .42, y + h * .35, x + w * .42, y + h * .46, fill=(60, 50, 40), r=w * .04, w=0)
    # cost
    d = w * .17
    drop(c, x - w * .36, y - h * .38, d)
    c.text(x - w * .36, y - h * .36, str(min(10, idx + 1)), d * 1.2, fill=WHITE, sw=max(1, d * .1))


def pin(c, x, y, s=1.0):
    if s < .05: return
    c.ell(x, y, 18 * s, 6 * s, fill=(0, 0, 0, ) if False else (60, 90, 60), w=0)
    c.poly([(x, y), (x - 22 * s, y - 52 * s), (x + 22 * s, y - 52 * s)], fill=RED, w=max(2, 4 * s))
    c.ell(x, y - 58 * s, 26 * s, 26 * s, fill=RED, w=max(2, 4 * s))
    c.poly([(x - 14 * s, y - 40 * s), (x + 14 * s, y - 40 * s), (x, y - 6 * s)], fill=RED, w=0)
    c.ell(x, y - 58 * s, 10 * s, 10 * s, fill=WHITE, w=max(2, 3 * s))


def clock(c, x, y, r, t, n=3):
    if r < 4: return
    c.ell(x, y, r, r, fill=WHITE, w=max(3, r * .06))
    for i in range(12):
        a = math.radians(i * 30 - 90)
        c.line([(x + r * .82 * math.cos(a), y + r * .82 * math.sin(a)), (x + r * .93 * math.cos(a), y + r * .93 * math.sin(a))], w=max(2, r * .04))
    k = int(t * n) % 12
    a = math.radians(k * 30 - 90)
    c.line([(x, y), (x + r * .6 * math.cos(a), y + r * .6 * math.sin(a))], fill=RED, w=max(3, r * .07))
    a2 = math.radians(((t * n / 12) % 1) * 360 - 90)
    c.line([(x, y), (x + r * .42 * math.cos(a2), y + r * .42 * math.sin(a2))], w=max(3, r * .08))
    c.ell(x, y, r * .06, r * .06, fill=OUT, w=0)

def limb(c, sx, sy, ang, wig, t, L, w, col, hand=SKIN, ph=0):
    a = math.radians(ang + wig * math.sin(t * 8 + ph))
    ex = sx + L * math.sin(a); ey = sy + L * math.cos(a)
    c.line([(sx, sy), (ex, ey)], fill=OUT, w=w + 8)
    c.line([(sx, sy), (ex, ey)], fill=col, w=w)
    c.ell(ex, ey, w * .62, w * .62, fill=hand, w=3)


def face(c, x, y, s, mood, t, look=(0, 0), skin=SKIN):
    lx, ly = look
    blink = (t % 3.1) < .1
    big = mood in ('shock', 'worry')
    rx = (16 if mood == 'shock' else 14 if big else 12.5) * s; ry = (20 if mood == 'shock' else 18 if big else 15) * s
    for sg in (-1, 1):
        ex = x + sg * 21 * s; ey = y - 6 * s
        if blink or mood == 'happy':
            if mood == 'happy' and not blink:
                c.arc(ex, ey + 4 * s, rx * .8, ry * .7, 200, 340, w=5 * s)
            else:
                c.line([(ex - rx * .8, ey + 2 * s), (ex + rx * .8, ey + 2 * s)], w=4 * s)
        else:
            c.ell(ex, ey, rx, ry, fill=WHITE, w=3.5 * s)
            pr = (3.6 if mood == 'shock' else 6.5) * s
            c.ell(ex + lx * 5 * s, ey + ly * 5 * s, pr, pr, fill=OUT, w=0)
            if mood in ('smug', 'sneaky'):
                c.rect(ex - rx - 2 * s, ey - ry - 3 * s, ex + rx + 2 * s, ey - 1 * s, fill=skin, r=0, w=0)
                c.line([(ex - rx, ey - 1 * s), (ex + rx, ey - 1 * s)], w=4 * s)
        # brows
        o = sg
        if mood == 'worry': a, b = (x + o * 36 * s, y - 24 * s), (x + o * 9 * s, y - 40 * s)
        elif mood == 'shock': a, b = (x + o * 34 * s, y - 40 * s), (x + o * 9 * s, y - 44 * s)
        elif mood in ('smug', 'sneaky'):
            a, b = ((x + o * 34 * s, y - 24 * s), (x + o * 8 * s, y - 24 * s)) if sg == -1 else ((x + o * 34 * s, y - 34 * s), (x + o * 8 * s, y - 40 * s))
        elif mood == 'flat': a, b = (x + o * 36 * s, y - 24 * s), (x + o * 6 * s, y - 28 * s)
        else: a, b = (x + o * 34 * s, y - 32 * s), (x + o * 9 * s, y - 34 * s)
        c.line([a, b], w=(10 if mood == 'flat' else 6) * s)
    my = y + 30 * s
    dark = (120, 30, 40)
    if mood in ('smile',): c.arc(x, my - 12 * s, 22 * s, 16 * s, 25, 155, w=5 * s)
    elif mood == 'happy':
        pts = [(x + 24 * s * math.cos(math.radians(a)), my - 6 * s + 24 * s * math.sin(math.radians(a))) for a in range(0, 181, 15)]
        c.poly(pts, fill=dark, w=4 * s)
        c.ell(x, my + 10 * s, 11 * s, 6 * s, fill=(255, 120, 130), w=0)
    elif mood == 'open': c.ell(x, my + 4 * s, 14 * s, 18 * s, fill=dark, w=4 * s)
    elif mood == 'shock': c.ell(x, my + 8 * s, 19 * s, 25 * s, fill=dark, w=4 * s)
    elif mood == 'worry':
        pts = [(x - 20 * s + i * 4 * s, my + 6 * s + math.sin(i * 1.4 + t * 14) * 3 * s) for i in range(11)]
        c.line(pts, w=5 * s)
        c.poly([(x + 54 * s, y - 34 * s), (x + 46 * s, y - 16 * s), (x + 62 * s, y - 16 * s)], fill=(120, 200, 255), w=2)
    elif mood in ('smug', 'sneaky'):
        c.line([(x - 18 * s, my + 2 * s), (x + 12 * s, my + 2 * s), (x + 26 * s, my - 8 * s)], w=5 * s)
    else: c.line([(x - 18 * s, my + 4 * s), (x + 18 * s, my + 4 * s)], w=5 * s)


def _hair(c, x, y, R, col, spikes=0, tall=0, inner=.62):
    pts = []
    n = max(spikes * 2, 24)
    for i in range(n + 1):
        a = math.radians(188 + 164 * i / n)
        rr = R * 1.02 + (tall if (spikes and i % 2 == 1) else 0)
        pts.append((x + rr * math.cos(a), y + rr * math.sin(a)))
    for i in range(n, -1, -1):
        a = math.radians(188 + 164 * i / n)
        rr = R * (inner + (.14 if i % 2 else 0))
        pts.append((x + rr * math.cos(a), y + rr * math.sin(a)))
    c.poly(pts, fill=col, w=4)


def _bob(t, s, sp=3): return math.sin(t * sp) * 3 * s


def dev(c, x, y, s=1.0, mood='smile', t=0, look=(0, 0), armR=None, armL=None, hat=False):
    y += _bob(t, s)
    shirt = (50, 110, 220)
    aR = armR or (28, 0); aL = armL or (-28, 0)
    c.rect(x - 56 * s, y + 52 * s, x + 56 * s, y + 175 * s, fill=shirt, r=20 * s, w=5 * s)
    c.poly([(x - 22 * s, y + 52 * s), (x + 22 * s, y + 52 * s), (x, y + 84 * s)], fill=WHITE, w=3 * s)
    c.poly([(x - 8 * s, y + 70 * s), (x + 8 * s, y + 70 * s), (x + 13 * s, y + 126 * s), (x, y + 144 * s), (x - 13 * s, y + 126 * s)], fill=RED, w=3 * s)
    limb(c, x + 50 * s, y + 78 * s, aR[0], aR[1], t, 86 * s, 22 * s, shirt, ph=0)
    limb(c, x - 50 * s, y + 78 * s, aL[0], aL[1], t, 86 * s, 22 * s, shirt, ph=2)
    c.ell(x, y, 58 * s, 58 * s, fill=SKIN, w=5 * s)
    _hair(c, x, y, 58 * s, (60, 40, 30), spikes=6, tall=26 * s)
    face(c, x, y, s, mood, t, look)
    if hat: chef_hat(c, x, y - 58 * s, s)

def tick(c, x, y, s=1.0, mood='smile', t=0, shades=False, armR=None, armL=None, halo=False):

    aR = armR or (30, 0); aL = armL or (-30, 0)

        c.ell(x + sg * 42 * s, y + 154 * s, 20 * s, 9 * s, fill=(240, 240, 250), w=4 * s)
    limb(c, x + 78 * s, y + 12 * s, aR[0], aR[1], t, 80 * s, 8 * s, OUT, hand=(240, 240, 250))
    limb(c, x - 78 * s, y + 12 * s, aL[0], aL[1], t, 80 * s, 8 * s, OUT, hand=(240, 240, 250), ph=2)
    c.rect(x - 78 * s, y - 100 * s, x + 78 * s, y + 100 * s, fill=(70, 78, 110), r=24 * s, w=6 * s)
    c.rect(x - 64 * s, y - 86 * s, x + 64 * s, y + 86 * s, fill=(60, 190, 230), r=14 * s, w=4 * s)
    c.ell(x, y + 112 * s, 8 * s, 8 * s, fill=(200, 205, 220), w=3 * s)
    ex = 26 * s; ey = y - 22 * s
    if shades:
        c.rect(x - 56 * s, ey - 16 * s, x + 56 * s, ey + 16 * s, fill=OUT, r=10 * s, w=0)
        c.line([(x - 8 * s, ey - 4 * s), (x + 8 * s, ey - 4 * s)], w=5 * s)
        c.line([(x - 44 * s, ey - 6 * s), (x - 32 * s, ey - 10 * s)], fill=WHITE, w=3 * s)
    else:
        for sg in (-1, 1):
            r = 15 * s if mood != 'worry' else 17 * s
            c.ell(x + sg * ex, ey, r, r * 1.15, fill=WHITE, w=4 * s)
            c.ell(x + sg * ex, ey + 2 * s, 7 * s, 7 * s, fill=OUT, w=0)
            if mood == 'worry': c.line([(x + sg * (ex + 14 * s), ey - 28 * s), (x + sg * (ex - 14 * s), ey - 40 * s)], w=5 * s)
    my = y + 34 * s
    if mood == 'grin':
        c.poly([(x + 36 * s * math.cos(math.radians(a)), my - 10 * s + 26 * s * math.sin(math.radians(a))) for a in range(0, 181, 15)], fill=WHITE, w=4 * s)
    elif mood == 'worry':
        c.line([(x - 26 * s + i * 5.2 * s, my + 8 * s + math.sin(i * 1.5 + t * 14) * 3 * s) for i in range(11)], w=5 * s)
    else:
        c.arc(x, my - 12 * s, 30 * s, 22 * s, 25, 155, w=6 * s)
    if halo: c.ell(x, y - 132 * s, 52 * s, 13 * s, fill=None, ol=YEL, w=9 * s)

        limb(c, x + 66 * s, y + 84 * s, 62, 0, t, 90 * s, 26 * s, suit, hand=skin)

        limb(c, x + 66 * s, y + 84 * s, 22, 0, t, 90 * s, 26 * s, suit, hand=skin)

def brain(c, x, y, s=1.0, t=0, mood='smile', band=False):

    body = (236, 112, 158); fold = (196, 70, 120)

    for dx, dy, rx, ry in parts: c.ell(x + dx * s, y + dy * s, rx * s + 4 * s, ry * s + 4 * s, fill=OUT, w=0)

    face(c, x, y + 14 * s, s * .85, mood, t, (0, 0), skin=body)
    if band:
        c.rect(x - 92 * s, y - 56 * s, x + 92 * s, y - 34 * s, fill=RED, r=8 * s, w=4 * s)
        c.poly([(x + 88 * s, y - 45 * s), (x + 126 * s, y - 70 * s), (x + 120 * s, y - 38 * s)], fill=RED, w=4 * s)
        c.poly([(x + 88 * s, y - 45 * s), (x + 124 * s, y - 24 * s), (x + 112 * s, y - 52 * s)], fill=RED, w=4 * s)


def giant(c, x, y, r, state='alive', t=0, hp=1.0, show_hp=True):
    dead = state == 'dead'
    g = (lambda col: gray(col)) if dead else (lambda col: col)
    ow = max(2, min(5, r * .05))
    y += 0 if dead else math.sin(t * 3) * r * .03
    skin = g((240, 190, 150)); tun = g((150, 95, 55)); hair = g((215, 120, 40))

def phone(c, x, y, w, h, t, mode='battle', seed=0, g=None, col=(25, 25, 35)):
    if w < 12 or h < 12: return
    c.rect(x - w / 2 - 5, y - h / 2 - 5, x + w / 2 + 5, y + h / 2 + 5, fill=OUT, r=w * .14, w=0)
    c.rect(x - w / 2, y - h / 2, x + w / 2, y + h / 2, fill=col, r=w * .13, w=3, ol=(150, 160, 195))
    sw = w * .84; sh = h * .90; sy = y + h * .01
    c.rect(x - sw / 2, sy - sh / 2, x + sw / 2, sy + sh / 2, fill=(30, 34, 50), r=w * .07, w=0)
    c.rect(x - w * .1, y - h / 2 + 4, x + w * .1, y - h / 2 + 9, fill=(70, 75, 95), r=3, w=0)
    x0, y0, x1, y1 = x - sw / 2, sy - sh / 2, x + sw / 2, sy + sh / 2
    if mode == 'spinner':
        c.rect(x0, y0, x1, y1, fill=(42, 46, 64), r=w * .06, w=0)
        c.arc(x, sy, sw * .2, sw * .2, t * 320, t * 320 + 270, fill=WHITE, w=max(4, sw * .05))
        return
    if mode == 'math':
        c.rect(x0, y0, x1, y1, fill=(20, 32, 58), r=w * .06, w=0)
        for i in range(1, 6): c.line([(x0 + 6, y0 + i * sh / 6), (x1 - 6, y0 + i * sh / 6)], fill=(36, 52, 88), w=2)
        return
    frozen = mode == 'frozen'
    c.rect(x0, y0, x1, y1, fill=(176, 208, 240) if frozen else (108, 184, 96), r=w * .06, w=0)
    ry = sy; rh = sh * .08
    c.rect(x0, ry - rh / 2, x1, ry + rh / 2, fill=(80, 150, 230), r=0, w=0)
    for bx in (x - sw * .25, x + sw * .25): c.rect(bx - sw * .07, ry - rh / 2 - 2, bx + sw * .07, ry + rh / 2 + 2, fill=(150, 110, 70), r=2, w=2)
    for ty, tc in ((y0 + sh * .1, RED), (y1 - sh * .1, BLUE)):
        for tx in (x - sw * .28, x + sw * .28):
            c.rect(tx - sw * .07, ty - sw * .07, tx + sw * .07, ty + sw * .07, fill=tc, r=3, w=2)
    if g is None and not frozen:
        for i in range(5):
            px = x + math.sin(t * 1.7 + i * 2.1 + seed * 3) * sw * .28
            py = sy + math.cos(t * 1.1 + i * 1.3 + seed * 1.7) * sh * .28
            rr = max(3, sw * .035)
            c.ell(px, py, rr, rr, fill=RED if py < sy else BLUE, w=2)
    if g is not None:
        giant(c, x, sy - sh * .06, sw * .15, g[0], t, 1.0, True)
    if frozen:
        for a in (20, 80, 140, 200, 260, 320):
            c.line([(x, sy), (x + sw * .5 * math.cos(math.radians(a)), sy + sh * .4 * math.sin(math.radians(a)))], fill=WHITE, w=3)
        wifi(c, x, y0 + sh * .04, max(.2, sw / 400), True)


def spiral_vortex(c, lt):
    c.d.rectangle([0, 0, W * S, H * S], fill=(14, 8, 40))
    cols = [(255, 60, 200), (60, 230, 255), (255, 230, 60), (140, 90, 255), (60, 255, 150)]
    cx, cy = 360, 430
    for arm in range(5):
        pts = []
        for i in range(1, 70):
            a = i * .16 + arm * 2 * math.pi / 5 + lt * 14
            r = i * i * .28 * (.5 + lt * 2.0)
            pts.append((cx + r * math.cos(a), cy + r * math.sin(a)))
        c.line(pts, fill=cols[arm], w=16)
